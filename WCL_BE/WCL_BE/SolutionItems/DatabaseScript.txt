CREATE PROCEDURE [account].[AddAccessToken]
	@AccountId bigint,
	@Token nvarchar(50)
AS
BEGIN
	INSERT INTO Token VALUES (@AccountId, @Token, DATEADD(mi, 45, GETDATE()))
	UPDATE Account SET LastLoginDate = GETDATE() WHERE Id = @AccountId
END
GO

CREATE PROCEDURE [account].[CheckToken]
	@Token nvarchar(50)
AS
BEGIN
	SELECT ISNULL(AccountId, 0) FROM Token WHERE Token = @Token AND ValidTo > GETDATE()
END
GO

CREATE PROCEDURE [account].[CreateNew]
	@Email nvarchar(255),
	@DisplayName nvarchar(25),
	@Password nvarchar(255),
	@Salt nvarchar(12),
	@GymName nvarchar(50),
	@CountryId bigint,
	@CityId bigint
AS
BEGIN
	
	DECLARE @NewAccount bigint
	INSERT INTO Account VALUES (@Email, @DisplayName, @Password, @Salt, NULL, NULL, NULL, GETDATE())
	
	SET @NewAccount = @@IDENTITY
	INSERT INTO Gym VALUES (@NewAccount, @GymName, 100000, 100000, @CountryId, @CityId, GETDATE(), 0)

END
GO

CREATE PROCEDURE [account].[ExtendAccessToken]
	@Token nvarchar(50)
AS
BEGIN
	UPDATE Token SET ValidTo = DATEADD(mi, 45, GETDATE()) WHERE Token = @Token
END
GO

CREATE PROCEDURE [account].[GetByEmail]
	@Email nvarchar(255)
AS
BEGIN
	SELECT TOP 1 * FROM Account WHERE LOWER(Email) = LOWER(@Email)
END
GO

CREATE PROCEDURE [account].[GetGymFromAccount]
	@AccountId bigint
AS
BEGIN
	SELECT TOP 1 Id FROM Gym WHERE AccountId = @AccountId AND Closed = 0
END
GO

CREATE PROCEDURE [fighter].[GetGymSpecificProspectCount]
	@GymId bigint
AS
BEGIN
	UPDATE Fighter SET Retired = GETDATE() WHERE GymId = @GymId AND Associated = 0 AND CreatedDate < DATEADD(dd, -3, GETDATE())
	SELECT COUNT(1) FROM Fighter WHERE GymId = @GymId
END
GO

CREATE PROCEDURE [fighter].[GetProspects]
	@GymId bigint
AS
BEGIN

	SELECT
		F.Id,
		F.FirstName,
		F.Surname,
		F.Nickname,
		W.Id AS [WeightclassId],
		W.Name AS [Weightclass],
		H.Id AS [HeightId],
		H.Name AS [Height],
		B.Id As [BackgroundId],
		B.Name As [Background],
		C.Id As [CityId],
		C.Name AS [City],
		CO.Id AS [CountryId],
		CO.Name AS [Country],
		F.Age,
		F.Chin,
		F.Heart,
		F.Strength,
		F.Agility,
		F.Stamina,
		F.Jabs,
		F.Crosses,
		F.Hooks,
		F.Uppercuts,
		F.Legkicks,
		F.Bodykicks,
		F.Headkicks,
		F.Backfists,
		F.Elbows,
		F.Kneestrikes,
		F.Takedowns,
		F.Clinch,
		F.TakedownDefence,
		F.HeadMovement,
		F.Footwork,
		F.Wrestling,
		F.Groundguard,
		F.Chokes,
		F.Armbars,
		F.Leglocks
	FROM
		Fighter F
	JOIN
		Weightclass W ON W.Id = F.WeightClassId
	JOIN
		Height H ON H.Id = F.HeightId
	JOIN
		Background B ON B.Id = F.BackgroundId
	JOIN
		City C ON C.Id = F.CityId
	JOIN
		Country CO ON CO.Id = F.CountryId
	WHERE
		Retired IS NULL AND 
		((GymId IS NOT NULL AND GymId = @GymId AND Associated = 0)
		OR
		(GymId IS NULL))
	ORDER BY
		F.Id

END
GO
